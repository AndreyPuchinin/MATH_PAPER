\section{Основные Определения}

\subsection{Структура, Форма, Метрика}

\begin{definition}[Структура]
    \textbf{Структура} — набор отношений между элементами, которые определяют их взаимодействие.  
    В теории групп — это операция, аксиомы, подгруппы.
\end{definition}

\begin{definition}[Форма]
    \textbf{Форма} — внешнее проявление структуры.  
    Пример: куб — форма, группа $O_h$ — структура.
\end{definition}

\begin{definition}[Метрика]
    \textbf{Метрика} — способ измерять расстояния.  
    В симметриях — движение, сохраняющее расстояние ($d(f(P), f(Q)) = d(P, Q)$).
\end{definition}

\subsection{Отображения и Симметрии}

\begin{definition}[Отображение]
    Правило $f: A \to B$, сопоставляющее каждому элементу из $A$ элемент из $B$.
\end{definition}

\begin{definition}[Движение (Isometry)]
    Отображение пространства на себя, сохраняющее расстояния: $d(f(P), f(Q)) = d(P, Q)$.
\end{definition}

\begin{definition}[Вращение (Rotation)]
    Движение, сохраняющее ориентацию. В $n$D — вокруг $(n-2)$-мерной плоскости.
\end{definition}

\begin{definition}[Отражение (Reflection)]
    Движение, меняющее ориентацию. Имеет неподвижную гиперплоскость.
\end{definition}

\begin{definition}[Симметрия (Symmetry)]
    Движение, переводящее фигуру в себя: $f(F) = F$.
\end{definition}

\begin{definition}[Собственная симметрия]
    Симметрия, сохраняющая ориентацию (вращения).
\end{definition}

\begin{definition}[Несобственная симметрия]
    Симметрия, меняющая ориентацию (отражения, композиции).
\end{definition}

\begin{definition}[Группа симметрий]
    Множество всех симметрий фигуры с операцией композиции.
\end{definition}

\section{Группа симметрий тессеракта (4D)}

\subsection{Общая формула}

Порядок группы симметрий $n$-мерного куба:
\[
|B_n| = 2^n \cdot n!
\]

Для $n=4$:
\[
|B_4| = 2^4 \cdot 4! = 16 \cdot 24 = 384
\]

\subsection{Подгруппы}

\begin{enumerate}
\item Циклическая подгруппа порядка 4: повороты вокруг одной оси на 90°, 180°, 270°, 360°.
    \item Подгруппа, изоморфная $S_3$: перестановки трёх координат, остальные фиксированы.
\end{enumerate}

\section{ANA и KATA: Геометрия Гиперкуба}

\begin{definition}[ANA]
    Внутренняя часть гиперкуба, соответствующая вершинам, где все координаты равны 1.
\end{definition}

\begin{definition}[KATA]
    Внешняя оболочка гиперкуба, соответствующая вершинам, где хотя бы одна координата равна 0.
\end{definition}
